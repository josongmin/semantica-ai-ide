## 3. Playwright Gate Runner (tests/gates/perf/spec.ts)

```typescript
import fs from "node:fs";
import path from "node:path";
import YAML from "yaml";
import { test, expect } from "@playwright/test";

type PerfGateSpec = any;

const SPEC_PATH = path.resolve(process.cwd(), "schemas/agent-ux/gates/performance.v1.yaml");  // RFC-006

test("gate-performance-v1", async ({ page }) => {
  const spec = YAML.parse(fs.readFileSync(SPEC_PATH, "utf-8")) as PerfGateSpec;

  await page.setViewportSize(spec.environment.viewport);

  for (const scenario of spec.scenarios) {
    await page.waitForTimeout(spec.environment.warmup_ms);

    await page.evaluate(() => (window as any).__perf?.reset?.());
    await runScenario(page, scenario);

    const snap = await page.evaluate(() => (window as any).__perf?.snapshot?.());
    const verdict = evaluate(spec.thresholds, snap);

    if (!verdict.pass) {
      const outDir = path.resolve(process.cwd(), "tests/gates/perf/out");
      fs.mkdirSync(outDir, { recursive: true });
      fs.writeFileSync(
        path.join(outDir, `${scenario.id}.json`),
        JSON.stringify({ scenario, snap, verdict }, null, 2),
        "utf-8"
      );
    }

    expect(verdict.pass, verdict.reason).toBeTruthy();
  }
});

async function runScenario(page: any, scenario: any) {
  for (const step of scenario.steps) {
    if (step.action === "open_view") {
      await page.evaluate((s: any) => (window as any).__agentUx?.open?.(s.view_id), step);
      continue;
    }
    if (step.action === "inject") {
      await page.evaluate((s: any) => (window as any).__agentUx?.inject?.(s), step);
      continue;
    }
    if (step.action === "stream") {
      await page.evaluate((s: any) => (window as any).__agentUx?.stream?.(s), step);
      continue;
    }
    if (step.action === "interact") {
      await page.evaluate((s: any) => (window as any).__agentUx?.interact?.(s), step);
      continue;
    }
    throw new Error(`unknown step.action: ${step.action}`);
  }
}

function evaluate(thresholds: any, snap: any) {
  const fails: string[] = [];

  const inpP95 = snap?.inp?.p95_ms ?? 0;
  const longCount = snap?.longTasks?.count ?? 0;
  const over16 = snap?.frames?.over16_ratio ?? 0;
  const rxPerSec = snap?.postMessage?.rx_per_sec ?? 0;
  const commitPerSec = snap?.uiCommit?.per_sec ?? 0;
  const heapGrowth = snap?.heap?.growth_mb ?? 0;

  if (inpP95 > thresholds.inp_p95_ms) fails.push(`INP p95 ${inpP95}ms > ${thresholds.inp_p95_ms}ms`);
  if (longCount > thresholds.long_task_count_max) fails.push(`LongTask count ${longCount} > ${thresholds.long_task_count_max}`);
  if (over16 > thresholds.frame_over_16ms_ratio_max) fails.push(`Frame over16 ratio ${over16} > ${thresholds.frame_over_16ms_ratio_max}`);
  if (rxPerSec > thresholds.postmessage_rx_per_sec_max) fails.push(`postMessage rx/s ${rxPerSec} > ${thresholds.postmessage_rx_per_sec_max}`);
  if (commitPerSec > thresholds.ui_commit_per_sec_max) fails.push(`UI commit/s ${commitPerSec} > ${thresholds.ui_commit_per_sec_max}`);
  if (heapGrowth > thresholds.heap_growth_mb_max) fails.push(`Heap growth ${heapGrowth}MB > ${thresholds.heap_growth_mb_max}MB`);

  return { pass: fails.length === 0, reason: fails.join(" | ") };
}
```

## 4. Webview Test Harness (src/ui/testHarness.ts)

### 4.1 설계

- Gate runner는 `page.evaluate(() => window.__agentUx.*())`로만 조작
- inject/stream은 실제 `window.postMessage(DELTA_BATCH)` 사용 → App의 message handler를 그대로 타서 postMessage rx/sec 계측이 의미 있게 나옴
- interact는 DOM 직접 조작 → INP 유사 측정 가능

### 4.2 API

| 메서드 | 용도 |
|--------|------|
| open(viewId) | view focus (webview 단독에서는 no-op) |
| inject(step) | messages / seed_message 주입 |
| stream(step) | append_text 스트리밍 |
| interact(step) | scroll / click_random_visible_rows / type_in_prompt |

### 4.3 핵심 구현

```typescript
// inject: postDeltaBatch로 DELTA_BATCH 전송
// stream: setInterval로 APPEND_TEXT 델타 16ms마다 전송
// interact: DOM querySelector로 스크롤/클릭/타이핑
```

`message-list-scroller` (data-testid), `message-row`, `#agent-ux-prompt` 필요.
